    <script>
        // DOM Elements
        const dropZone = document.getElementById('drop-zone');
        const fileInput = document.getElementById('file-input');
        const conversionTypeSelect = document.getElementById('conversion-type');
        
        const uiStateUpload = document.getElementById('ui-state-upload');
        const uiStateProgress = document.getElementById('ui-state-progress');
        const uiStateError = document.getElementById('ui-state-error');
        const uiStateSuccess = document.getElementById('ui-state-success');
        
        const selectedFileInfo = document.getElementById('selected-file-info');
        const selectedFilename = document.getElementById('selected-filename');
        const selectedFilesize = document.getElementById('selected-filesize');
        const btnRemoveFile = document.getElementById('btn-remove-file');
        const btnSubmit = document.getElementById('btn-submit');
        
        const progressBar = document.getElementById('progress-bar');
        const progressPercentage = document.getElementById('progress-percentage');
        const progressBytes = document.getElementById('progress-bytes');
        const progressStatusText = document.getElementById('progress-status-text');
        const progressFilename = document.getElementById('progress-filename');
        const serverProcessingIndicator = document.getElementById('server-processing-indicator');
        const errorMessage = document.getElementById('error-message');

        // State
        let currentFile = null;
        const API_ENDPOINT = 'https://pdf-backend-server.onrender.com/upload';

        // Map types to expected output extensions for fallback naming
        const extensionMap = {
            'pdf-to-word': '.docx',
            'image-to-word': '.docx',
            'image-to-pdf': '.pdf',
            'word-to-excel': '.xlsx',
            'ppt-to-pdf': '.pdf',
            'word-to-pdf': '.pdf'
        };

        ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(eventName => {
            dropZone.addEventListener(eventName, preventDefaults, false);
        });

        function preventDefaults(e) {
            e.preventDefault();
            e.stopPropagation();
        }

        ['dragenter', 'dragover'].forEach(eventName => {
            dropZone.addEventListener(eventName, () => dropZone.classList.add('drag-active'), false);
        });

        ['dragleave', 'drop'].forEach(eventName => {
            dropZone.addEventListener(eventName, () => dropZone.classList.remove('drag-active'), false);
        });

        dropZone.addEventListener('drop', (e) => {
            if(e.dataTransfer.files.length) {
                handleFile(e.dataTransfer.files[0]);
            }
        });

        fileInput.addEventListener('change', (e) => {
            if(e.target.files.length) {
                handleFile(e.target.files[0]);
            }
        });

        btnRemoveFile.addEventListener('click', () => {
            currentFile = null;
            fileInput.value = '';
            selectedFileInfo.classList.add('hidden');
            dropZone.classList.remove('hidden');
            btnSubmit.disabled = true;
        });

        function handleFile(file) {
            currentFile = file;
            
            // Format file size
            let sizeStr = '';
            if (file.size < 1024) sizeStr = file.size + ' B';
            else if (file.size < 1048576) sizeStr = (file.size / 1024).toFixed(1) + ' KB';
            else sizeStr = (file.size / 1048576).toFixed(2) + ' MB';
            
            selectedFilename.textContent = file.name;
            selectedFilesize.textContent = sizeStr;
            
            dropZone.classList.add('hidden');
            selectedFileInfo.classList.remove('hidden');
            btnSubmit.disabled = false;
        }

        function formatBytes(bytes, decimals = 2) {
            if (!+bytes) return '0 Bytes';
            const k = 1024;
            const dm = decimals < 0 ? 0 : decimals;
            const sizes = ['B', 'KB', 'MB', 'GB'];
            const i = Math.floor(Math.log(bytes) / Math.log(k));
            return `${parseFloat((bytes / Math.pow(k, i)).toFixed(dm))} ${sizes[i]}`;
        }

        btnSubmit.addEventListener('click', () => {
            if (!currentFile) return;

            const conversionType = conversionTypeSelect.value;
            
            // Switch UI to progress
            uiStateUpload.classList.add('hidden-animated');
            uiStateProgress.classList.remove('hidden');
            uiStateProgress.classList.add('flex');
            
            progressFilename.textContent = currentFile.name;
            progressBar.style.width = '0%';
            progressPercentage.textContent = '0%';
            progressBytes.textContent = `0 B / ${formatBytes(currentFile.size)}`;
            progressStatusText.textContent = 'Uploading to server...';
            serverProcessingIndicator.classList.add('hidden');

            // Setup FormData
            const formData = new FormData();
            formData.append('file', currentFile);
            formData.append('conversionType', conversionType);

            // Setup XMLHttpRequest to track real progress
            const xhr = new XMLHttpRequest();
            xhr.open('POST', API_ENDPOINT, true);
            xhr.responseType = 'blob'; // Expecting a file back

            // Track Upload Progress
            xhr.upload.onprogress = (event) => {
                if (event.lengthComputable) {
                    const percentComplete = Math.round((event.loaded / event.total) * 100);
                    progressBar.style.width = percentComplete + '%';
                    progressPercentage.textContent = percentComplete + '%';
                    progressBytes.textContent = `${formatBytes(event.loaded)} / ${formatBytes(event.total)}`;
                    
                    if (percentComplete === 100) {
                        progressStatusText.textContent = 'Upload complete.';
                        serverProcessingIndicator.classList.remove('hidden');
                    }
                }
            };

            xhr.onload = function() {
                if (xhr.status >= 200 && xhr.status < 300) {
                    // Success! Process the downloaded blob
                    const blob = xhr.response;
                    
                    // Try to get filename from Content-Disposition header
                    let filename = '';
                    const disposition = xhr.getResponseHeader('Content-Disposition');
                    if (disposition && disposition.indexOf('attachment') !== -1) {
                        const filenameRegex = /filename[^;=\n]*=((['"]).*?\2|[^;\n]*)/;
                        const matches = filenameRegex.exec(disposition);
                        if (matches != null && matches[1]) { 
                            filename = matches[1].replace(/['"]/g, '');
                        }
                    }

                    // Fallback filename if header is missing
                    if (!filename) {
                        const originalName = currentFile.name.substring(0, currentFile.name.lastIndexOf('.')) || currentFile.name;
                        const ext = extensionMap[conversionType];
                        filename = `${originalName}_converted${ext}`;
                    }

                    // Trigger actual download
                    const url = window.URL.createObjectURL(blob);
                    const a = document.createElement('a');
                    a.style.display = 'none';
                    a.href = url;
                    a.download = filename;
                    document.body.appendChild(a);
                    a.click();
                    
                    // Cleanup
                    window.URL.revokeObjectURL(url);
                    document.body.removeChild(a);

                    // Show Success UI
                    uiStateProgress.classList.add('hidden-animated');
                    uiStateProgress.classList.remove('flex');
                    uiStateSuccess.classList.remove('hidden');
                    uiStateSuccess.classList.add('flex');
                } else {
                    // Handle Server Error
                    let errStr = `Server responded with status ${xhr.status}.`;
                    
                    // Attempt to read blob as text to show specific server error if they sent JSON
                    if (xhr.response && xhr.response.type === 'application/json') {
                        const reader = new FileReader();
                        reader.onload = function() {
                            try {
                                const json = JSON.parse(reader.result);
                                if(json.error) errStr = json.error;
                                else if(json.message) errStr = json.message;
                            } catch(e) {}
                            showError(errStr);
                        };
                        reader.readAsText(xhr.response);
                    } else {
                        showError(errStr);
                    }
                }
            };

            xhr.onerror = function() {
                // Network Error (CORS, server down, etc.)
                showError('Network error. Ensure your backend is running at localhost:3000 and CORS is enabled.');
                document.getElementById('server-status-indicator').classList.replace('text-slate-300', 'text-error');
            };

            // Send the request
            xhr.send(formData);
        });

        function showError(msg) {
            errorMessage.textContent = msg;
            uiStateProgress.classList.add('hidden-animated');
            uiStateProgress.classList.remove('flex');
            uiStateError.classList.remove('hidden');
            uiStateError.classList.add('flex');
        }

        function resetUI() {
            currentFile = null;
            fileInput.value = '';
            
            uiStateError.classList.add('hidden-animated');
            uiStateError.classList.remove('flex');
            uiStateSuccess.classList.add('hidden-animated');
            uiStateSuccess.classList.remove('flex');
            
            selectedFileInfo.classList.add('hidden');
            dropZone.classList.remove('hidden');
            
            uiStateUpload.classList.remove('hidden-animated');
            btnSubmit.disabled = true;
            document.getElementById('server-status-indicator').classList.replace('text-error', 'text-slate-300');
        }

    </script>
